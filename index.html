export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') return res.status(200).end(); try { let params; if (typeof req.body === 'string') { try { const decoded = Buffer.from(req.body, 'base64').toString('utf-8'); params = JSON.parse(decoded); } catch (e) { params = JSON.parse(req.body); } } else { params = req.body; } const { k, m, c } = params; const url = `https://generativelanguage.googleapis.com/v1beta/models/${m}:generateContent?key=${k}`; const response = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: c }) }); const resText = await response.text(); const outB64 = Buffer.from(resText, 'utf-8').toString('base64'); return res.status(200).json({ d: outB64 }); } catch (err) { return res.status(200).json({ err: err.toString() }); } }
